    class BggThing < ApplicationRecord
      has_many :bgg_collections, dependent: :destroy
      has_many :users, through: :bgg_collections

      validates :objectid, presence: true, uniqueness: true
      validates :name, presence: true

      # D-15: Collection merge — takes an array of user_ids and returns deduplicated things
      # that at least one of the specified users owns.
      scope :merged_for_users, ->(user_ids) {
        joins(:bgg_collections)
          .where(bgg_collections: { user_id: user_ids })
          .distinct
      }

      # D-16: Enrich merged results with owner names for frontend display
      # Uses GROUP_CONCAT for SQLite, would be array_agg for Postgres.
      # Phase 3 GraphQL resolves owners per thing; this scope provides the raw data.
      scope :with_owner_names, -> {
        joins(:bgg_collections)
          .joins("INNER JOIN users ON users.id = bgg_collections.user_id")
          .select("bgg_things.*, GROUP_CONCAT(users.username, ',') AS owner_usernames")
          .group("bgg_things.id")
      }
    end

    **D-14, D-16:** Create `app/services/bgg/collection_persister.rb`:

    ```ruby
    # frozen_string_literal: true

    class Bgg::CollectionPersister
      attr_reader :username, :collection_data

      def initialize(username, collection_data)
        @username = username
        @collection_data = collection_data
      end

      # Persists the normalized collection data for a user.
      # Creates or updates BggThing records for each item, then creates/replaces
      # BggCollection records linking the user to the things they own.
      #
      # Returns the User record (creating it as a provisional profile if needed,
      # with username as the display name — Phase 4 will add account creation).
      def persist!
        user = find_or_create_user!
        items = collection_data[:items]

        ActiveRecord::Base.transaction do
          # Clear existing collections for this user before re-importing
          user.bgg_collections.destroy_all

          items.each do |item|
            thing = upsert_thing!(item)
            create_collection_entry!(user, thing, item)
          end
        end

        user.reload
      end

      private

      def find_or_create_user!
        User.find_or_create_by!(username: username) do |u|
          # Provision minimal profile — just the BGG username for now
          # Phase 4 adds email/password accounts with username binding
        end
      rescue ActiveRecord::RecordInvalid => e
        raise Bgg::InvalidUsernameError, "Failed to create user profile: #{e.message}"
      end

      def upsert_thing!(item)
        thing = BggThing.find_or_initialize_by(objectid: item[:object_id].to_i)
        thing.update!(thing_attributes(item))
        thing
      end

      def create_collection_entry!(user, thing, item)
        user.bgg_collections.create!(
          bgg_thing: thing,
          quantity: 1,
          comment: item[:comment].to_s,
          is_own: item.dig(:status, :own) || false,
          is_previously_owned: item.dig(:status, :prev_owned) || false,
          is_for_trade: item.dig(:status, :for_trade) || false,
          is_want: item.dig(:status, :want) || false,
          is_want_to_play: item.dig(:status, :want_to_play) || false,
          is_want_to_buy: item.dig(:status, :want_to_buy) || false,
          is_wishlist: item.dig(:status, :wishlist) || false,
          wishlist_priority: nil  # BGG collection API doesn't provide this in the standard response
        )
      end

      def thing_attributes(item)
        stats = item[:stats] || {}
        {
          name: item[:name].to_s,
          year_published: item[:year_published],
          min_players: stats[:min_players],
          max_players: stats[:max_players],
          min_playtime: stats[:min_playtime],
          max_playtime: stats[:max_playtime],
          complexity: stats[:average_weight],
          thumbnail: item[:thumbnail].to_s,
          image: item[:image].to_s,
          description: item[:description].to_s,
          is_expansion: item[:subtype] == "boardgameexpansion" || item[:is_expansion] || false,
          bgg_type: item[:subtype].to_s,
          raw_data: item  # Store the full normalized hash as JSONB safety net
        }
      end
    end
    ```

    Key design points:
    - **Transactional import**: Entire import in a single transaction. If any record fails, nothing is persisted (D-11 idempotency requirement).
    - **Clear-and-rebuild**: Existing BggCollection rows for the user are deleted before re-import. This ensures the collection always reflects the live BGG state. The user's `bgg_collections.destroy_all` is inside the transaction so failure rolls back deletions too.
    - **Upsert by BGG ID**: `BggThing.find_or_initialize_by(objectid:)` ensures we don't create duplicate things when two users both own the same game. The database unique index on `objectid` provides the second line of defense.
    - **Provisional user**: If the User record doesn't already exist, `find_or_create_by!` creates one with the BGG username as the display name. This supports the "preview collections without an account" flow that Phase 3 needs. Phase 4 upgrades these to full accounts.
    - **Raw_data safety net**: The full normalized hash is stored as JSONB under `raw_data`. If Phase 3 needs a field we didn't explicitly map (e.g., `year_published` on collection items vs thing items), it can be extracted from `raw_data` without a re-fetch from BGG (per RESEARCH.md Recommendation 3).
